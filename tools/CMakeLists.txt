find_package(tinyxml2 QUIET)
if(NOT tinyxml2_FOUND)
  # Ubuntu jammy 未提供 CMake config 时退回 pkg-config/手查
  find_path(TINYXML2_INCLUDE_DIR tinyxml2.h)
  find_library(TINYXML2_LIBRARY tinyxml2)
  if(NOT TINYXML2_INCLUDE_DIR OR NOT TINYXML2_LIBRARY)
    message(WARNING "tinyxml2 not found: nebula doc subcommand disabled")
    add_executable(nebula nebula.cpp)
    target_compile_definitions(nebula PRIVATE NEBULABT_NO_DOC)
    target_link_libraries(nebula PRIVATE nebulabt::nebulabt)
    return()
  endif()
  add_library(tinyxml2_shim INTERFACE)
  target_include_directories(tinyxml2_shim INTERFACE ${TINYXML2_INCLUDE_DIR})
  target_link_libraries(tinyxml2_shim INTERFACE ${TINYXML2_LIBRARY})
  add_library(tinyxml2::tinyxml2 ALIAS tinyxml2_shim)
endif()

add_executable(nebula nebula.cpp)
target_link_libraries(nebula PRIVATE nebulabt::nebulabt tinyxml2::tinyxml2)
